\section{Tujuan}

\begin{enumerate}
    \item Menganalisis sentimen terhadap destinasi wisata berdasarkan ulasan Google Reviews menggunakan model IndoBERT.
    
    \item Mengidentifikasi aspek atau topik utama yang dibahas wisatawan menggunakan metode BERTopic.
    
    \item Memberikan visualisasi sebagai bahan evaluasi bagi pengelola destinasi wisata.
\end{enumerate}
